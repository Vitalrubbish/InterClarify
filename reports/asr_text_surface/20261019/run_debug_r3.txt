#!/usr/bin/env bash
set -euo pipefail
PROBE=/hpc_stor03/sjtu_home/xuan.zhang/interclarify-layer1-training-20261019/data/probes/asr-text-surface-20261019
SOURCE=/hpc_stor03/sjtu_home/xuan.zhang/interclarify-layer1-training-20261019/source/07b0d51
ASR_ENV=/hpc_stor03/sjtu_home/xuan.zhang/interclarify-english-integration-20261018-v1/envs/asr
MODEL=/hpc_stor03/sjtu_home/xuan.zhang/interclarify-asr-selection/models/nemotron_en
OUT="$PROBE/runs/debug-r3"
MONITOR=''
cleanup() {
  rc=$?
  if [ -n "$MONITOR" ]; then kill "$MONITOR" 2>/dev/null || true; fi
  printf '%s\n' "$rc" > "$OUT/exit_code.txt"
}
trap cleanup EXIT
export PYTHONNOUSERSITE=1 HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1 TOKENIZERS_PARALLELISM=false
export OMP_NUM_THREADS=4 MKL_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4 CUDA_VISIBLE_DEVICES=0
ulimit -v 33554432
ulimit -f 1048576
cd "$SOURCE"
git rev-parse HEAD > "$OUT/source_commit.txt"
nvidia-smi --query-gpu=index,name,uuid,memory.total,memory.used --format=csv > "$OUT/gpu_before.csv"
nvidia-smi --query-gpu=index,memory.used,utilization.gpu --format=csv -l 1 > "$OUT/gpu_samples.csv" &
MONITOR=$!
taskset -c 0-3 nice -n 10 timeout --signal=TERM --kill-after=30 600 "$ASR_ENV/bin/python" -u scripts/remote/probe_asr_text_surface.py --manifest "$PROBE/manifest.json" --model "$MODEL" --output "$OUT/report.json" > "$OUT/probe.log" 2> "$OUT/stderr.log"
